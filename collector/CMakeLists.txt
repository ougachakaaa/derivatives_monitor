cmake_minimum_required(VERSION 3.16)
project(derivatives_collector LANGUAGES CXX)
include(../experiments/ctp/common/ctp_sdk.cmake)
include(CTest)
find_package(SQLite3 3.24 REQUIRED)
add_library(collector_storage src/storage.cpp)
target_compile_features(collector_storage PUBLIC cxx_std_17)
target_compile_options(collector_storage PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(collector_storage PUBLIC src "${CTP_COMMON_DIR}")
target_link_libraries(collector_storage PUBLIC SQLite::SQLite3 ctp_md_sdk Threads::Threads)
add_executable(data_collector src/main.cpp)
configure_ctp_executable(data_collector)
add_library(collector_runtime src/session.cpp src/runtime.cpp)
target_compile_options(collector_runtime PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(collector_runtime PUBLIC collector_storage)
target_link_libraries(data_collector PRIVATE collector_runtime)
if(BUILD_TESTING)
    add_executable(config_test tests/config_test.cpp)
    configure_ctp_executable(config_test)
    target_include_directories(config_test PRIVATE src tests)
    add_test(NAME config COMMAND config_test "${CMAKE_CURRENT_SOURCE_DIR}/config.example.ini")
    add_executable(state_test tests/state_test.cpp)
    configure_ctp_executable(state_test)
    target_include_directories(state_test PRIVATE src tests)
    add_test(NAME state COMMAND state_test)
    add_executable(storage_test tests/storage_test.cpp)
    configure_ctp_executable(storage_test)
    target_include_directories(storage_test PRIVATE src tests)
    target_link_libraries(storage_test PRIVATE collector_storage)
    add_test(NAME storage COMMAND storage_test)
    add_executable(runtime_test tests/runtime_test.cpp)
    configure_ctp_executable(runtime_test)
    target_include_directories(runtime_test PRIVATE src tests)
    target_link_libraries(runtime_test PRIVATE collector_runtime)
    add_test(NAME runtime COMMAND runtime_test)
    add_executable(health_test tests/health_test.cpp)
    configure_ctp_executable(health_test)
    target_include_directories(health_test PRIVATE src tests)
    target_link_libraries(health_test PRIVATE collector_runtime)
    add_test(NAME health COMMAND health_test)
    # Only tests preload a synthetic SDK; the production target always links the official SDK.
    if(CTP_EFFECTIVE_LINK_MODE STREQUAL "SHARED")
        add_library(fake_md SHARED tests/fake_sdk.cpp)
        target_compile_features(fake_md PRIVATE cxx_std_17)
        target_compile_options(fake_md PRIVATE -Wall -Wextra -Wpedantic)
        target_include_directories(fake_md PRIVATE src tests "${CTP_COMMON_DIR}")
        target_include_directories(fake_md SYSTEM PRIVATE "${CTP_SDK_ROOT}")
        target_link_libraries(fake_md PRIVATE Threads::Threads)
        add_executable(process_test tests/process_test.cpp)
        target_compile_features(process_test PRIVATE cxx_std_17)
        target_compile_options(process_test PRIVATE -Wall -Wextra -Wpedantic)
        target_link_libraries(process_test PRIVATE SQLite::SQLite3 Threads::Threads)
        add_test(NAME process COMMAND process_test "$<TARGET_FILE:data_collector>" "$<TARGET_FILE:fake_md>")
        set_tests_properties(process PROPERTIES TIMEOUT 25)
    endif()
endif()
